import heapq # Priority Queue = Min-Heap queue = [] def add_patient(severity, name): heapq.heappush(queue, (severity, name)) # O(log n) def treat_next(): if queue: severity, name = heapq.heappop(queue) return name # Most critical first! # O(log n) — heap rebalance hota hai # heappush: O(log n) # heappop: O(log n) # Min peek: O(1) via queue[0]

class Node: def __init__(self, id, name): self.id = id self.name = name self.left = None self.right = None def search(root, id): if not root or root.id == id: return root if id < root.id: return search(root.left, id) return search(root.right, id) # Time: O(log n) average

Shortest Path
Python BFS Code
from collections import deque def bfs_path(graph, start, end): queue = deque([(start, [start])]) visited = {start} while queue: node, path = queue.popleft() if node == end: return path for neighbor in graph[node]: if neighbor not in visited: visited.add(neighbor) queue.append( (neighbor, path + [neighbor]) ) # Time: O(V + E) # Guaranteed shortest path!

Python Stack Code
# Python list as Stack stack = [] def book(patient, doctor, date): appointment = { 'patient': patient, 'doctor': doctor, 'date': date } stack.append(appointment) # PUSH O(1) def undo(): if stack: return stack.pop() # POP O(1) # LIFO = Last In, First Out # Last booked = first to undo!
